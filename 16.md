# Multi-Stock Data
tickers = ['TSM', 'NVDA', 'INTC', 'AMD']
data = yf.download(tickers, period="3y")['Close']
print(data.head())

# 收盤價比較 
data.plot(figsize=(12, 6))
plt.title('Stock Prices (Raw)')
plt.show()

# 正規化比較
norm_data = (data / data.iloc[0]) * 100
norm_data.plot(figsize=(12, 6))
plt.title('Normalized Performance (Base=100)')
plt.show()

# 相關性熱圖
returns = data.pct_change().dropna()
corr = returns.corr()
sns.heatmap(corr, annot=True, cmap='coolwarm', vmin=-1, vmax=1)
plt.title('Correlation Matrix')
plt.show()

# 散佈圖比較
sns.scatterplot(x=returns['TSM'], y=returns['NVDA'])
plt.title('TSM vs NVDA Daily Returns')
plt.show()

# 散佈圖矩陣
sns.pairplot(returns)
plt.show()

# 風險分析
volatility = returns.std() * (252 ** 0.5)
volatility.plot.bar(color=['red', 'green', 'blue', 'orange'])
plt.title('Annualized Volatility (Risk)')
plt.show()

# 總報酬分析 
total_return = (data.iloc[-1] / data.iloc[0]) - 1
total_return.plot.bar()
plt.title('Total Return (3 Years)')
plt.show()

# 回撤分析
running_max = df['Close'].cummax()
drawdown = (df['Close'] / running_max) - 1
plt.figure(figsize=(12, 4))
plt.fill_between(drawdown.index, drawdown, color='red', alpha=0.3)
plt.title('Drawdown (Decline from Peak)')
plt.show()

# K線圖
簡易版: 紅綠柱狀圖代表漲跌
subset = df[-60:]
colors = ['red' if c > o else 'green' for c, o in zip(subset['Close'], subset['Open'])]
plt.bar(subset.index, subset['Close'] - subset['Open'], bottom=subset['Open'], color=colors)
plt.title('Simple Candlestick (Last 60 Days)')
plt.show()

# 布林通道
df['MA20'] = df['Close'].rolling(window=20).mean()
df['Std'] = df['Close'].rolling(20).std()
df['Upper'] = df['MA20'] + 2*df['Std']
df['Lower'] = df['MA20'] - 2*df['Std']

sub = df[-200:]
plt.figure(figsize=(12, 6))
plt.plot(sub['Close'], label='Price')
plt.plot(sub['Upper'], label='Upper', linestyle='--')
plt.plot(sub['Lower'], label='Lower', linestyle='--')
plt.fill_between(sub.index, sub['Upper'], sub['Lower'], alpha=0.1)
plt.title('Bollinger Bands')
plt.show()
