import yfinance as yf
import pandas as pd
import matplotlib.pyplot as plt
import seaborn as sns
解決中文顯示問題 (if needed) & 風格
sns.set_theme(style="darkgrid")

# Fetch Data
stock = yf.Ticker("2330.TW")
df = stock.history(period="5y")
print(df.head())

收盤價走勢
plt.figure(figsize=(12, 6))
plt.plot(df.index, df['Close'], label='Close Price')
plt.title('TSMC (2330.TW) Stock Price')
plt.xlabel('Date')
plt.ylabel('Price (TWD)')
plt.legend()
plt.show()

# 交易量
plt.figure(figsize=(12, 4))
plt.bar(df.index, df['Volume'], color='red', alpha=0.5)
plt.title('Trading Volume')
plt.show()

# 均線策略
df['MA20'] = df['Close'].rolling(window=20).mean()
df['MA60'] = df['Close'].rolling(window=60).mean()

plt.figure(figsize=(12, 6))
plt.plot(df['Close'], label='Close', alpha=0.5)
plt.plot(df['MA20'], label='MA20 (Month)', color='orange')
plt.plot(df['MA60'], label='MA60 (Quarter)', color='green')
plt.legend()
plt.title('Moving Averages')
plt.show()

# 日收益率
df['Return'] = df['Close'].pct_change()
plt.figure(figsize=(12, 4))
plt.plot(df['Return'], linestyle='-', linewidth=0.5)
plt.title('Daily Returns (Volatility)')
plt.show()

# 收益率分佈
sns.histplot(df['Return'].dropna(), bins=50, kde=True)
plt.title('Return Distribution')
plt.show()

# 收益率箱型圖
sns.boxplot(x=df['Return'])
plt.title('Daily Return Boxplot')
plt.show()

# 累積報酬率
df['CumReturn'] = (1 + df['Return']).cumprod()
plt.figure(figsize=(12, 6))
plt.fill_between(df.index, df['CumReturn'], alpha=0.4)
plt.plot(df['CumReturn'])
plt.title('Cumulative Return (Growth of $1)')
plt.show()
